javascriptconst SHEET_ID = '1sF-haW0mrnwSpl6Ck3gWWt4501gbmi-GIPn475qNYqI';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const action = data.action;
    const ss = SpreadsheetApp.openById(SHEET_ID);

    if (action === 'saveBill') {
      const sheet = ss.getSheetByName('Bill_Items');
      data.items.forEach(item => {
        sheet.appendRow([
          data.billNo, data.date, data.customer, data.mkt,
          item.itemName, item.qty, item.rate, item.amount,
          item.transport, item.bags, data.lrNo
        ]);
      });
      return ContentService.createTextOutput(JSON.stringify({status:'success'})).setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'savePayment') {
      const sheet = ss.getSheetByName('Payments');
      sheet.appendRow([data.date, data.customer, data.amount, data.note]);
      return ContentService.createTextOutput(JSON.stringify({status:'success'})).setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'saveCustomer') {
      const sheet = ss.getSheetByName('Customers');
      sheet.appendRow([data.name, data.openingBalance]);
      return ContentService.createTextOutput(JSON.stringify({status:'success'})).setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'saveItem') {
      const sheet = ss.getSheetByName('Items');
      sheet.appendRow([data.itemName, data.defaultRate]);
      return ContentService.createTextOutput(JSON.stringify({status:'success'})).setMimeType(ContentService.MimeType.JSON);
    }

  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status:'error', message:err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    const action = e.parameter.action;
    const ss = SpreadsheetApp.openById(SHEET_ID);

    if (action === 'getCustomers') {
      const data = ss.getSheetByName('Customers').getDataRange().getValues();
      return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'getItems') {
      const data = ss.getSheetByName('Items').getDataRange().getValues();
      return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'getBillItems') {
      const data = ss.getSheetByName('Bill_Items').getDataRange().getValues();
      return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'getPayments') {
      const data = ss.getSheetByName('Payments').getDataRange().getValues();
      return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
    }

  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status:'error', message:err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
